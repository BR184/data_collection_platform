```sql
CREATE INDEX CONCURRENTLY idx_issue_fact_testing_phase_scope

ON issue_fact(reason_category) WHERE deleted = false;

-- 复合索引优化组合查询
CREATE INDEX CONCURRENTLY idx_issue_fact_composite_stats
ON issue_fact(deleted, testing_phase, severity_level, is_excluded);

-- GIN索引优化模块名数组查询
CREATE INDEX CONCURRENTLY idx_issue_fact_module_gin
ON issue_fact USING gin(string_to_array(module_names, ', '))
WHERE deleted = false;

-- 优化镜像表查询
CREATE INDEX CONCURRENTLY idx_labels_project_title ON labels(project_id, title);
CREATE INDEX CONCURRENTLY idx_issues_project_updated ON issues(project_id, updated_at DESC);
CREATE INDEX CONCURRENTLY idx_merge_requests_project_updated ON merge_requests(target_project_id, updated_at DESC);

-- 优化常见筛选组合
CREATE INDEX CONCURRENTLY idx_issue_fact_project_phase
ON issue_fact(project_id, testing_phase, deleted, is_excluded) WHERE deleted = false;

CREATE INDEX CONCURRENTLY idx_issue_fact_severity_fixed
ON issue_fact(severity_level, is_fixed)
WHERE deleted = false AND is_excluded = false;
```

## 修复原理

### 为什么CrownCAD-issues-29373没有统计到？

**老平台逻辑**：
```java
// CC_PRODUCT项目（ID=325）不排除"功能屏蔽"、"已拒绝"、"建议"
if (!StringUtils.equals(projectId, ProjectList.CC_PRODUCT_ID.getProjectId())) {
    query.and(wrapper -> wrapper
            .notLike("category", "功能屏蔽")
            .notLike("bug_status", "已拒绝")
            .notLike("category", "建议")
    );
}
```

**新平台旧逻辑**：
```java
// 所有项目都排除"功能屏蔽"、"已拒绝"、"建议"
for (String label : EXCLUDED_LABELS) {
    if (IssueRuleSupport.hasLabel(labels, label)) {
        return label;  // 直接排除
    }
}
```

**结论**：
- 如果issue-29373带有"建议"标签或其他排除标签
- 老平台（CrownCAD项目）：不排除，正常统计
- 新平台（旧逻辑）：排除，不统计
- 新平台（新逻辑）：不排除，正常统计 ✅
